        // Global Helper: Custom Backoff Fetch Wrapper for Gemini API
        const apiKey = ""; // Implicitly injected in live Canvas environment
        async function callGeminiAPI(payload, modelName = "gemini-3-flash-preview") {
            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/${modelName}:generateContent?key=${apiKey}`;
            
            let retries = 3;
            let delay = 1000;
            while (retries > 0) {
                try {
                    const response = await fetch(apiUrl, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(payload)
                    });
                    if (response.ok) {
                        return await response.json();
                    }
                    if (response.status === 429) {
                        // Rate limit - Apply backoff
                        await new Promise(resolve => setTimeout(resolve, delay));
                        delay *= 2;
                        retries--;
                    } else {
                        throw new Error(`HTTP Error: ${response.status}`);
                    }
                } catch (error) {
                    if (retries === 1) throw error;
                    await new Promise(resolve => setTimeout(resolve, delay));
                    delay *= 2;
                    retries--;
                }
            }
        }

        // Base64 helper converters (mandatory inline base64 encoders/decoders)
        function base64ToArrayBuffer(base64) {
            const binaryString = window.atob(base64);
            const len = binaryString.length;
            const bytes = new Uint8Array(len);
            for (let i = 0; i < len; i++) {
                bytes[i] = binaryString.charCodeAt(i);
            }
            return bytes.buffer;
        }

        function pcmToWav(pcm16Data, sampleRate) {
            const buffer = new ArrayBuffer(44 + pcm16Data.byteLength);
            const view = new DataView(buffer);

            // "RIFF" chunk descriptor
            writeString(view, 0, 'RIFF');
            view.setUint32(4, 36 + pcm16Data.byteLength, true);
            writeString(view, 8, 'WAVE');

            // "fmt " sub-chunk
            writeString(view, 12, 'fmt ');
            view.setUint32(16, 16, true); // Size of fmt chunk
            view.setUint16(20, 1, true);   // Uncompressed PCM format
            view.setUint16(22, 1, true);   // Mono channel
            view.setUint32(24, sampleRate, true);
            view.setUint32(28, sampleRate * 2, true); // Byte rate (sampleRate * 1 channel * 16 bits/8)
            view.setUint16(32, 2, true);              // Block align
            view.setUint16(34, 16, true);             // 16 bits per sample

            // "data" sub-chunk
            writeString(view, 36, 'data');
            view.setUint32(40, pcm16Data.byteLength, true);

            // Copy PCM bytes directly
            const pcmBytes = new Uint8Array(pcm16Data);
            for (let i = 0; i < pcmBytes.length; i++) {
                view.setUint8(44 + i, pcmBytes[i]);
            }

            return new Blob([view], { type: 'audio/wav' });
        }

        function writeString(view, offset, string) {
            for (let i = 0; i < string.length; i++) {
                view.setUint8(offset + i, string.charCodeAt(i));
            }
        }

        async function generateAIAffirmation() {
            const card = document.getElementById('aff-player-card');
            const txt = document.getElementById('aff-text');
            const playBtn = document.getElementById('btn-play-tts');
            
            // Clean active audio instances
            if (audioInstance) {
                audioInstance.pause();
                audioInstance = null;
                document.getElementById('play-icon').className = "fa-solid fa-play text-lg ml-1";
                document.getElementById('tts-audio-indicator').classList.add('hidden');
            }

            card.classList.remove('hidden');
            txt.innerHTML = `<i class="fa-solid fa-spinner animate-spin"></i> Mindfully crafting text affirmation...`;
            playBtn.setAttribute('disabled', 'true');

            // 1. Generate Calm Affirmation Text
            const textPrompt = `Generate a single, deeply calming, elegant mental affirmation (under 25 words) for a person who feels extremely ${selectedMood}. Address them with grace, warmth, and mindfulness. No hashtags or intros.`;
            const textPayload = {
                contents: [{ parts: [{ text: textPrompt }] }],
                systemInstruction: { parts: [{ text: "You are a thoughtful spiritual guide. Write brief, tranquil affirmations." }] }
            };

            try {
                const textResult = await callGeminiAPI(textPayload);
                generatedAffirmationText = textResult.candidates?.[0]?.content?.parts?.[0]?.text || "Breathe, you are safe and supported.";
                txt.textContent = `"${generatedAffirmationText}"`;

                // 2. Synthesize with Gemini Speech TTS model
                txt.innerHTML = `"${generatedAffirmationText}" <span class="block text-xs text-gold-600 mt-2"><i class="fa-solid fa-spinner animate-spin"></i> Harmonizing spoken AI audio...</span>`;
                const selectedVoice = document.getElementById('aff-voice').value;

                const ttsPayload = {
                    contents: [{
                        parts: [{ text: `Say with extreme peace, slow pacing, and gentle reassurance: ${generatedAffirmationText}` }]
                    }],
                    generationConfig: {
                        responseModalities: ["AUDIO"],
                        speechConfig: {
                            voiceConfig: {
                                prebuiltVoiceConfig: { voiceName: selectedVoice }
                            }
                        }
                    }
                };

                const ttsResponse = await callGeminiAPI(ttsPayload, "gemini-2.5-flash-preview-tts");
                const part = ttsResponse?.candidates?.[0]?.content?.parts?.[0];
                const audioData = part?.inlineData?.data;
                const mimeType = part?.inlineData?.mimeType;

                if (audioData && mimeType && mimeType.startsWith("audio/")) {
                    const sampleRate = parseInt(mimeType.match(/rate=(\d+)/)?.[1] || "24000", 10);
                    const pcmBuffer = base64ToArrayBuffer(audioData);
                    const pcm16 = new Int16Array(pcmBuffer);
                    const wavBlob = pcmToWav(pcm16.buffer, sampleRate);
                    generatedAudioUrl = URL.createObjectURL(wavBlob);

                    // Re-enable Player controls
                    txt.textContent = `"${generatedAffirmationText}"`;
                    playBtn.removeAttribute('disabled');
                } else {
                    throw new Error("TTS conversion response missing payload.");
                }

            } catch (err) {
                txt.textContent = `"${generatedAffirmationText}"`;
                txt.innerHTML += `<span class="block text-xs text-red-600 mt-2"><i class="fa-solid fa-triangle-exclamation"></i> Audio failed to stream, but enjoy the written affirmation!</span>`;
            }
        }

        function playAffirmationTTS() {
            const icon = document.getElementById('play-icon');
            const indicator = document.getElementById('tts-audio-indicator');

            if (!generatedAudioUrl) return;

            if (audioInstance && !audioInstance.paused) {
                audioInstance.pause();
                icon.className = "fa-solid fa-play text-lg ml-1";
                indicator.classList.add('hidden');
            } else {
                if (!audioInstance) {
                    audioInstance = new Audio(generatedAudioUrl);
                    audioInstance.addEventListener('ended', () => {
                        icon.className = "fa-solid fa-play text-lg ml-1";
                        indicator.classList.add('hidden');
                        audioInstance = null;
                    });
                }
                audioInstance.play();
                icon.className = "fa-solid fa-pause text-lg";
                indicator.classList.remove('hidden');
            }
        }
